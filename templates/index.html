Developer API

Programmatic Integration

curl -X POST https://snip-it.me/api/shorten \
  -H "X-API-Key: YOUR_API_KEY" \
  -H "Content-Type: application/json" \
  -d '{"url": "https://example.com/very-long-url", "alias": "my-promo"}'
import requests

resp = requests.post(
    "https://snip-it.me/api/shorten",
    headers={"X-API-Key": "YOUR_API_KEY"},
    json={"url": "https://example.com/very-long-url", "alias": "my-promo"}
)
print(resp.json())  # {"code": "my-promo", "short_url": "https://snip-it.me/my-promo"}
